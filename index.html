""" update_rathafandhoo_portal.py Safely updates `index.html` in the current working directory from a source HTML file (or stamps a build/version marker into an existing `index.html`), optionally creates a timestamped backup, and can build `my-rathafandhoo-portal.zip`. Usage: python update_rathafandhoo_portal.py python update_rathafandhoo_portal.py --source code.html --zip """ from __future__ import annotations import argparse from datetime import datetime, timezone import re import shutil import sys import zipfile from pathlib import Path DEFAULT_README = """# My Rathafandhoo (މައި ރަތަފަންދޫ) — Portal Package A bilingual (Dhivehi RTL in Faruma font & English LTR in Inter/Plus Jakarta Sans) community portal for **GDh. Rathafandhoo**, Huvadhu Atoll, Maldives. """ def parse_args() -> argparse.Namespace: """Parse command-line arguments.""" parser = argparse.ArgumentParser( description="Update index.html for the Rathafandhoo portal cleanly." ) parser.add_argument( "--source", type=Path, default=None, help="Path to a DIFFERENT HTML file to copy into index.html (e.g., code.html).", ) parser.add_argument( "--target", type=Path, default=Path("index.html"), help="Target HTML file path (default: index.html).", ) parser.add_argument( "--force-stamp", action="store_true", help="Explicitly stamp a new build timestamp into the existing target file without a --source file.", ) parser.add_argument( "--no-backup", action="store_true", help="Skip creating a backup copy before modifying index.html.", ) parser.add_argument( "--zip", action="store_true", help="Also generate my-rathafandhoo-portal.zip after updating index.html.", ) return parser.parse_args() def create_backup(target_path: Path) -> Path | None: """Create a timestamped backup of an existing target file.""" if not target_path.exists() or not target_path.is_file(): return None stamp = datetime.now(timezone.utc).strftime("%Y%m%d-%H%M%S") backup_path = target_path.with_name(f"{target_path.stem}.backup-{stamp}{target_path.suffix}") shutil.copy2(target_path, backup_path) return backup_path def inject_build_stamp(html_content: str, build_label: str) -> str: """Insert or update a visible build stamp comment and footer badge in HTML.""" comment_line = f"" if re.search(r"", html_content): html_content = re.sub( r"", comment_line, html_content, count=1, ) else: html_content = f"{comment_line}\n{html_content}" badge_pattern = r']*>.*?' badge_html = ( f'' f'Build: {build_label}' ) if re.search(badge_pattern, html_content, flags=re.DOTALL): return re.sub(badge_pattern, badge_html, html_content, count=1, flags=re.DOTALL) if "" in html_content: return html_content.replace("", f"
{badge_html}
\n ", 1) return html_content def write_zip_archive(html_path: Path, zip_path: Path) -> Path: """Package the updated HTML file and README into a ZIP file.""" with zipfile.ZipFile(zip_path, mode="w", compression=zipfile.ZIP_DEFLATED) as zf: zf.write(html_path, arcname="index.html") zf.writestr("README.md", DEFAULT_README) return zip_path.resolve() def main() -> int: """Run the portal update workflow with clear error handling.""" args = parse_args() target_path = args.target try: if args.source is not None: if not args.source.is_file(): raise FileNotFoundError(f"Source file not found: {args.source}") if args.source.resolve() == target_path.resolve(): raise ValueError( f"--source ('{args.source}') and --target ('{target_path}') are the same file; " "pass a different source file instead." ) raw_html = args.source.read_text(encoding="utf-8") else: if not target_path.is_file(): raise FileNotFoundError( f"Target '{target_path}' does not exist. Pass --source to create it." ) if not args.force_stamp: raise ValueError( f"No --source file provided and --force-stamp was not set. " f"Refusing to overwrite '{target_path}' with its own existing contents. " "Pass --source new_page.html or use --force-stamp." ) raw_html = target_path.read_text(encoding="utf-8") if not raw_html.strip(): raise ValueError("HTML content is empty; aborting to avoid writing a blank page.") if not args.no_backup and target_path.exists(): backup_file = create_backup(target_path) if backup_file: print(f"Backup created: {backup_file}") build_label = datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC") updated_html = inject_build_stamp(raw_html, build_label) target_path.write_text(updated_html, encoding="utf-8") print(f"Updated '{target_path}' successfully ({build_label}).") if args.zip: zip_file = write_zip_archive(target_path, Path("my-rathafandhoo-portal.zip")) print(f"Created archive: {zip_file}") return 0 except Exception as exc: print(f"Error: {exc}", file=sys.stderr) return 1 if __name__ == "__main__": sys.exit(main())
